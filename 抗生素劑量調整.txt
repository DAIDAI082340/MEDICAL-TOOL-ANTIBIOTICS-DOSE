// 初始化 DOM 與搜尋事件
const tableBody = document.getElementById("tableBody");
const searchInput = document.getElementById("searchInput");
const drugSelect = document.getElementById("drugSelect");
const filterBtns = document.querySelectorAll(".filter-btn");
const resultCount = document.getElementById("resultCount");
const featuredCard = document.getElementById("featuredCard");

let currentRoute = "ALL";
let flatDrugList = [];

drugDatabase.forEach(cat => {
    cat.drugs.forEach(d => {
        flatDrugList.push(d);
        const opt = document.createElement("option");
        opt.value = d.name;
        opt.textContent = `${d.name} [${d.route}]`;
        drugSelect.appendChild(opt);
    });
});

function getRouteBadge(route) {
    if (route.includes("IV") && route.includes("PO")) return `<span class="route-badge route-mix">${route}</span>`;
    if (route.includes("IV")) return `<span class="route-badge route-iv">${route}</span>`;
    if (route.includes("PO")) return `<span class="route-badge route-po">${route}</span>`;
    if (route.includes("Inh")) return `<span class="route-badge route-inh">${route}</span>`;
    return `<span class="route-badge">${route}</span>`;
}

function renderTable(searchTerm = "", routeFilter = "ALL") {
    tableBody.innerHTML = "";
    let totalMatches = 0;
    const term = searchTerm.trim().toLowerCase();

    drugDatabase.forEach(cat => {
        const filteredDrugs = cat.drugs.filter(d => {
            const matchText = (d.name + " " + d.note + " " + d.norm + " " + d.mod + " " + d.sev).toLowerCase();
            const matchesSearch = !term || matchText.includes(term);
            const matchesRoute = (routeFilter === "ALL") || d.route.includes(routeFilter);
            return matchesSearch && matchesRoute;
        });

        if (filteredDrugs.length > 0) {
            const catTr = document.createElement("tr");
            catTr.className = "category-row";
            catTr.innerHTML = `<td colspan="7">【${cat.category}】</td>`;
            tableBody.appendChild(catTr);

            filteredDrugs.forEach(d => {
                totalMatches++;
                const tr = document.createElement("tr");
                tr.className = "drug-row";
                
                let displayName = d.name;
                if (term) {
                    const regex = new RegExp(`(${term})`, "gi");
                    displayName = displayName.replace(regex, `<span class="highlight-match">$1</span>`);
                }

                tr.innerHTML = `
                    <td class="drug-name">${displayName}</td>
                    <td style="text-align:center;">${getRouteBadge(d.route)}</td>
                    <td class="dose-cell">${d.stat.replace(/\n/g, '<br>')}</td>
                    <td class="dose-cell dose-normal">${d.norm.replace(/\n/g, '<br>')}</td>
                    <td class="dose-cell dose-moderate">${d.mod.replace(/\n/g, '<br>')}</td>
                    <td class="dose-cell dose-severe">${d.sev.replace(/\n/g, '<br>')}</td>
                    <td class="notes-cell">${d.note}</td>
                `;
                tableBody.appendChild(tr);
            });
        }
    });

    resultCount.textContent = `共顯示 ${totalMatches} 種抗生素品項`;
}

searchInput.addEventListener("input", (e) => {
    renderTable(e.target.value, currentRoute);
});

filterBtns.forEach(btn => {
    btn.addEventListener("click", () => {
        filterBtns.forEach(b => b.classList.remove("active"));
        btn.classList.add("active");
        currentRoute = btn.getAttribute("data-route");
        renderTable(searchInput.value, currentRoute);
    });
});

drugSelect.addEventListener("change", (e) => {
    const selectedName = e.target.value;
    if (!selectedName) {
        featuredCard.classList.remove("active");
        return;
    }
    const drug = flatDrugList.find(d => d.name === selectedName);
    if (drug) {
        document.getElementById("featName").textContent = drug.name;
        document.getElementById("featRoute").outerHTML = getRouteBadge(drug.route);
        document.getElementById("featStat").innerHTML = drug.stat.replace(/\n/g, '<br>') || "—";
        document.getElementById("featNorm").innerHTML = drug.norm.replace(/\n/g, '<br>') || "—";
        document.getElementById("featMod").innerHTML = drug.mod.replace(/\n/g, '<br>') || "—";
        document.getElementById("featSev").innerHTML = drug.sev.replace(/\n/g, '<br>') || "—";
        document.getElementById("featNote").textContent = drug.note || "無特別備註";
        featuredCard.classList.add("active");
        featuredCard.scrollIntoView({ behavior: 'smooth', block: 'start' });
    }
});

renderTable();
